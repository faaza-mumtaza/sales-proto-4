     1→"use client";
     2→
     3→// Admin: kelola FAQ (tambah/edit/hapus/urutan/tampil-sembunyi).
     4→
     5→import { useMemo, useState, type FormEvent } from "react";
     6→import { useQuery, useQueryClient } from "@tanstack/react-query";
     7→import { HelpCircle, Trash2, Search, Plus, Pencil, Eye, EyeOff, ArrowUp, ArrowDown } from "lucide-react";
     8→import { toast } from "sonner";
     9→import {
    10→  Dialog,
    11→  DialogContent,
    12→  DialogHeader,
    13→  DialogTitle,
    14→  DialogFooter,
    15→} from "@/components/ui/dialog";
    16→import {
    17→  Select,
    18→  SelectContent,
    19→  SelectItem,
    20→  SelectTrigger,
    21→  SelectValue,
    22→} from "@/components/ui/select";
    23→import { Accordion, AccordionContent, AccordionItem, AccordionTrigger } from "@/components/ui/accordion";
    24→import { apiGet, apiPost, apiPut, apiPatch, apiDelete } from "@/lib/api";
    25→import { usePageMeta } from "@/lib/router";
    26→import { useSelection } from "@/lib/use-selection";
    27→import { BulkActionBar } from "@/components/admin/bulk-action-bar";
    28→import { AdminShell } from "./admin-shell";
    29→import { FAQ_KATEGORI_LABEL } from "@/lib/site-utils";
    30→
    31→interface FaqAdmin {
    32→  id: string;
    33→  kategori: string;
    34→  pertanyaan: string;
    35→  jawaban: string;
    36→  urutan: number;
    37→  is_published: boolean;
    38→  created_at: string;
    39→  updated_at: string;
    40→}
    41→
    42→const KATEGORI_WARNA: Record<string, string> = {
    43→  umum: "bg-suzuki-navy/10 text-suzuki-navy",
    44→  pembelian: "bg-suzuki-red/10 text-suzuki-red",
    45→  purnajual: "bg-green-100 text-green-700",
    46→};
    47→
    48→interface FormState {
    49→  id?: string;
    50→  kategori: string;
    51→  pertanyaan: string;
    52→  jawaban: string;
    53→  urutan: number;
    54→  is_published: boolean;
    55→}
    56→
    57→const EMPTY_FORM: FormState = {
    58→  kategori: "umum",
    59→  pertanyaan: "",
    60→  jawaban: "",
    61→  urutan: 0,
    62→  is_published: true,
    63→};
    64→
    65→export function AdminFaqView() {
    66→  usePageMeta("FAQ — Admin Suzuki BSB");
    67→  const qc = useQueryClient();
    68→  const [search, setSearch] = useState("");
    69→  const [filterKategori, setFilterKategori] = useState<string>("all");
    70→  const [dialogOpen, setDialogOpen] = useState(false);
    71→  const [form, setForm] = useState<FormState>(EMPTY_FORM);
    72→  const [saving, setSaving] = useState(false);
    73→  const [error, setError] = useState<string | null>(null);
    74→  const [bulkBusy, setBulkBusy] = useState(false);
    75→
    76→  const { data, isLoading, isError, refetch } = useQuery({
    77→    queryKey: ["admin", "faqs"],
    78→    queryFn: () => apiGet<{ faqs: FaqAdmin[]; counts: { published: number; hidden: number } }>(
    79→      "/api/admin/faqs",
    80→    ),
    81→  });
    82→
    83→  const faqs = data?.faqs ?? [];
    84→  const counts = data?.counts;
    85→
    86→  const filtered = useMemo(() => {
    87→    const q = search.trim().toLowerCase();
    88→    return faqs.filter((f) => {
    89→      const cocokKategori = filterKategori === "all" || f.kategori === filterKategori;
    90→      const cocokCari =
    91→        q === "" ||
    92→        f.pertanyaan.toLowerCase().includes(q) ||
    93→        f.jawaban.toLowerCase().includes(q);
    94→      return cocokKategori && cocokCari;
    95→    });
    96→  }, [faqs, search, filterKategori]);
    97→
    98→  // Daftar final yang tampil (filter kartu ringkasan tampil/sembunyi + urutan)
    99→  // — dipakai juga untuk seleksi aksi massal agar konsisten dengan yang terlihat.
   100→  const visible = useMemo(() => {
   101→    let list = filtered;
   102→    if (filterKategori === "__published__") list = list.filter((f) => f.is_published);
   103→    if (filterKategori === "__hidden__") list = list.filter((f) => !f.is_published);
   104→    return [...list].sort((a, b) => a.urutan - b.urutan || a.created_at.localeCompare(b.created_at));
   105→  }, [filtered, filterKategori]);
   106→
   107→  const sortedAll = useMemo(
   108→    () => [...faqs].sort((a, b) => a.urutan - b.urutan || a.created_at.localeCompare(b.created_at)),
   109→    [faqs],
   110→  );
   111→
   112→  // Seleksi item untuk aksi massal
   113→  const sel = useSelection(visible);
   114→
   115→  async function bulkPublish(isPublished: boolean) {
   116→    if (sel.count === 0) return;
   117→    setBulkBusy(true);
   118→    try {
   119→      const res = await apiPatch<{ updated: number }>("/api/admin/faqs", {
   120→        ids: [...sel.selected],
   121→        is_published: isPublished,
   122→      });
   123→      await qc.invalidateQueries({ queryKey: ["admin", "faqs"] });
   124→      await qc.invalidateQueries({ queryKey: ["admin", "stats"] });
   125→      await qc.invalidateQueries({ queryKey: ["faqs", "public"] });
   126→      toast.success(`${res.updated} FAQ ${isPublished ? "ditampilkan" : "disembunyikan"}`);
   127→      sel.clear();
   128→    } catch (e) {
   129→      toast.error(e instanceof Error ? e.message : "Gagal mengubah status FAQ secara massal");
   130→    } finally {
   131→      setBulkBusy(false);
   132→    }
   133→  }
   134→
   135→  async function bulkDelete() {
   136→    if (sel.count === 0) return;
   137→    if (!window.confirm(`Hapus ${sel.count} FAQ terpilih? Tindakan ini tidak bisa dibatalkan.`)) return;
   138→    setBulkBusy(true);
   139→    try {
   140→      const res = await apiDelete<{ deleted: number }>("/api/admin/faqs", {
   141→        ids: [...sel.selected].join(","),
   142→      });
   143→      await qc.invalidateQueries({ queryKey: ["admin", "faqs"] });
   144→      await qc.invalidateQueries({ queryKey: ["admin", "stats"] });
   145→      await qc.invalidateQueries({ queryKey: ["faqs", "public"] });
   146→      toast.success(`${res.deleted} FAQ dihapus`);
   147→      sel.clear();
   148→    } catch (e) {
   149→      toast.error(e instanceof Error ? e.message : "Gagal menghapus FAQ secara massal");
   150→    } finally {
   151→      setBulkBusy(false);
   152→    }
   153→  }
   154→
   155→  function openCreate() {
   156→    const urutanBaru = faqs.length > 0 ? Math.max(...faqs.map((f) => f.urutan)) + 1 : 1;
   157→    setForm({ ...EMPTY_FORM, urutan: urutanBaru });
   158→    setError(null);
   159→    setDialogOpen(true);
   160→  }
   161→
   162→  function openEdit(f: FaqAdmin) {
   163→    setForm({
   164→      id: f.id,
   165→      kategori: f.kategori,
   166→      pertanyaan: f.pertanyaan,
   167→      jawaban: f.jawaban,
   168→      urutan: f.urutan,
   169→      is_published: f.is_published,
   170→    });
   171→    setError(null);
   172→    setDialogOpen(true);
   173→  }
   174→
   175→  async function onSubmit(e: FormEvent) {
   176→    e.preventDefault();
   177→    setError(null);
   178→
   179→    if (form.pertanyaan.trim().length < 8) {
   180→      setError("Pertanyaan minimal 8 karakter.");
   181→      return;
   182→    }
   183→    if (form.jawaban.trim().length < 10) {
   184→      setError("Jawaban minimal 10 karakter.");
   185→      return;
   186→    }
   187→
   188→    setSaving(true);
   189→    try {
   190→      const payload = {
   191→        ...(form.id ? { id: form.id } : {}),
   192→        kategori: form.kategori,
   193→        pertanyaan: form.pertanyaan.trim(),
   194→        jawaban: form.jawaban.trim(),
   195→        urutan: form.urutan,
   196→        is_published: form.is_published,
   197→      };
   198→      if (form.id) {
   199→        await apiPut("/api/admin/faqs", payload);
   200→        toast.success("FAQ diperbarui");
   201→      } else {
   202→        await apiPost("/api/admin/faqs", payload);
   203→        toast.success("FAQ ditambahkan");
   204→      }
   205→      setDialogOpen(false);
   206→      await qc.invalidateQueries({ queryKey: ["admin", "faqs"] });
   207→      await qc.invalidateQueries({ queryKey: ["faqs", "public"] });
   208→    } catch (err) {
   209→      const msg = err instanceof Error ? err.message : "Gagal menyimpan FAQ";
   210→      setError(msg);
   211→    } finally {
   212→      setSaving(false);
   213→    }
   214→  }
   215→
   216→  async function onToggle(f: FaqAdmin) {
   217→    try {
   218→      await apiPatch("/api/admin/faqs", { id: f.id, is_published: !f.is_published });
   219→      await qc.invalidateQueries({ queryKey: ["admin", "faqs"] });
   220→      await qc.invalidateQueries({ queryKey: ["faqs", "public"] });
   221→      toast.success(f.is_published ? "FAQ disembunyikan dari website" : "FAQ ditampilkan di website");
   222→    } catch (e) {
   223→      toast.error(e instanceof Error ? e.message : "Gagal mengubah status");
   224→    }
   225→  }
   226→
   227→  async function onDelete(f: FaqAdmin) {
   228→    if (!window.confirm(`Hapus FAQ "${f.pertanyaan}" secara permanen?`)) return;
   229→    try {
   230→      await apiDelete("/api/admin/faqs", { id: f.id });
   231→      await qc.invalidateQueries({ queryKey: ["admin", "faqs"] });
   232→      await qc.invalidateQueries({ queryKey: ["faqs", "public"] });
   233→      toast.success("FAQ dihapus");
   234→    } catch (e) {
   235→      toast.error(e instanceof Error ? e.message : "Gagal menghapus FAQ");
   236→    }
   237→  }
   238→
   239→  /** Tukar urutan dua FAQ bersebelahan pada daftar terurut, lalu simpan. */
   240→  async function onMove(f: FaqAdmin, arah: -1 | 1) {
   241→    const urutList = [...faqs].sort((a, b) => a.urutan - b.urutan || a.created_at.localeCompare(b.created_at));
   242→    const idx = urutList.findIndex((x) => x.id === f.id);
   243→    const tukar = urutList[idx + arah];
   244→    if (!tukar) return;
   245→    try {
   246→      // Tukar nilai urutan kedua item
   247→      await apiPut("/api/admin/faqs", {
   248→        id: f.id,
   249→        kategori: f.kategori,
   250→        pertanyaan: f.pertanyaan,
   251→        jawaban: f.jawaban,
   252→        urutan: tukar.urutan,
   253→        is_published: f.is_published,
   254→      });
   255→      await apiPut("/api/admin/faqs", {
   256→        id: tukar.id,
   257→        kategori: tukar.kategori,
   258→        pertanyaan: tukar.pertanyaan,
   259→        jawaban: tukar.jawaban,
   260→        urutan: f.urutan,
   261→        is_published: tukar.is_published,
   262→      });
   263→      await qc.invalidateQueries({ queryKey: ["admin", "faqs"] });
   264→      await qc.invalidateQueries({ queryKey: ["faqs", "public"] });
   265→    } catch (e) {
   266→      toast.error(e instanceof Error ? e.message : "Gagal mengubah urutan");
   267→    }
   268→  }
   269→
   270→  return (
   271→    <AdminShell>
   272→      <div className="space-y-6">
   273→        <div className="flex flex-wrap items-start justify-between gap-4">
   274→          <div>
   275→            <h1 className="text-2xl font-bold text-suzuki-navy">FAQ (Pertanyaan Umum)</h1>
   276→            <p className="text-muted-foreground text-sm mt-1">
   277→              Kelola daftar pertanyaan umum yang tampil di halaman Kontak website.
   278→            </p>
   279→          </div>
   280→          <button
   281→            onClick={openCreate}
   282→            className="inline-flex items-center gap-2 bg-suzuki-red hover:bg-suzuki-red/90 text-white font-semibold px-4 py-2.5 rounded-lg text-sm transition-all hover:shadow-lg hover:shadow-suzuki-red/30 active:scale-95"
   283→          >
   284→            <Plus className="w-4 h-4" aria-hidden />
   285→            Tambah FAQ
   286→          </button>
   287→        </div>
   288→
   289→        {/* Ringkasan */}
   290→        <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   291→          {[
   292→            { id: "all", label: "Total", n: faqs.length },
   293→            { id: "published", label: "Tampil di website", n: counts?.published ?? 0 },
   294→            { id: "hidden", label: "Disembunyikan", n: counts?.hidden ?? 0 },
   295→          ].map((s) => (
   296→            <button
   297→              key={s.id}
   298→              onClick={() =>
   299→                setFilterKategori(
   300→                  s.id === "all"
   301→                    ? "all"
   302→                    : s.id === "published"
   303→                      ? "__published__"
   304→                      : "__hidden__",
   305→                )
   306→              }
   307→              className={`rounded-xl border p-4 text-left transition-all ${
   308→                (s.id === "all" && filterKategori === "all") ||
   309→                (s.id === "published" && filterKategori === "__published__") ||
   310→                (s.id === "hidden" && filterKategori === "__hidden__")
   311→                  ? "border-suzuki-red/50 bg-white shadow-sm ring-1 ring-suzuki-red/30"
   312→                  : "border-border bg-white/60 hover:border-suzuki-navy/30"
   313→              }`}
   314→              aria-pressed={s.id === "all" ? filterKategori === "all" : s.id === "published" ? filterKategori === "__published__" : filterKategori === "__hidden__"}
   315→            >
   316→              <p className="text-2xl font-bold text-suzuki-navy">{s.n}</p>
   317→              <p className="text-xs text-muted-foreground mt-0.5">{s.label}</p>
   318→            </button>
   319→          ))}
   320→        </div>
   321→
   322→        {/* Filter bar */}
   323→        <div className="flex flex-col sm:flex-row sm:items-center gap-3">
   324→          <div className="relative flex-1 sm:max-w-xs">
   325→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" aria-hidden />
   326→            <input
   327→              value={search}
   328→              onChange={(e) => setSearch(e.target.value)}
   329→              placeholder="Cari pertanyaan / jawaban…"
   330→              aria-label="Cari FAQ"
   331→              className="w-full pl-9 pr-3 py-2 border rounded-lg text-sm bg-white focus:outline-none focus:ring-2 focus:ring-suzuki-red/50"
   332→            />
   333→          </div>
   334→          <div className="flex flex-wrap gap-1.5" role="group" aria-label="Filter kategori FAQ">
   335→            {["all", "umum", "pembelian", "purnajual"].map((k) => (
   336→              <button
   337→                key={k}
   338→                onClick={() => setFilterKategori(k === "all" ? "all" : k)}
   339→                aria-pressed={filterKategori === k || (k === "all" && filterKategori === "all")}
   340→                className={`px-3.5 py-1.5 rounded-full text-xs font-medium border transition-all active:scale-95 ${
   341→                  (k === "all" && filterKategori === "all") || filterKategori === k
   342→                    ? "bg-suzuki-navy text-white border-suzuki-navy"
   343→                    : "bg-white text-muted-foreground border-border hover:border-suzuki-navy/40 hover:text-suzuki-navy"
   344→                }`}
   345→              >
   346→                {k === "all" ? "Semua Kategori" : (FAQ_KATEGORI_LABEL[k] ?? k)}
   347→              </button>
   348→            ))}
   349→          </div>
   350→          <p className="text-sm text-muted-foreground sm:ml-auto" aria-live="polite">
   351→            {filtered.length} FAQ
   352→          </p>
   353→        </div>
   354→
   355→        {/* Bilah aksi massal — tampil saat ada item terpilih */}
   356→        <BulkActionBar
   357→          count={sel.count}
   358→          total={visible.length}
   359→          onSelectAll={sel.selectAll}
   360→          onClear={sel.clear}
   361→          statuses={[
   362→            { value: "true", label: "Tampilkan" },
   363→            { value: "false", label: "Sembunyikan" },
   364→          ]}
   365→          onBulkStatus={(s) => void bulkPublish(s === "true")}
   366→          onBulkDelete={() => void bulkDelete()}
   367→          busy={bulkBusy}
   368→        />
   369→
   370→        {/* Daftar */}
   371→        {isLoading ? (
   372→          <div className="bg-white rounded-xl border border-border p-8 text-center text-muted-foreground text-sm">
   373→            Memuat FAQ…
   374→          </div>
   375→        ) : isError ? (
   376→          <div className="bg-white rounded-xl border border-border p-8 text-center">
   377→            <p className="text-muted-foreground text-sm mb-2">Gagal memuat FAQ.</p>
   378→            <button onClick={() => void refetch()} className="text-suzuki-red underline text-sm">
   379→              Coba lagi
   380→            </button>
   381→          </div>
   382→        ) : visible.length === 0 ? (
   383→          <div className="bg-white rounded-xl border border-border p-10 text-center">
   384→            <HelpCircle className="w-10 h-10 text-muted-foreground/40 mx-auto mb-3" aria-hidden />
   385→            <p className="text-muted-foreground text-sm">
   386→              {faqs.length === 0
   387→                ? "Belum ada FAQ. Klik \"Tambah FAQ\" untuk membuat pertanyaan pertama."
   388→                : "Tidak ada FAQ yang cocok dengan filter."}
   389→            </p>
   390→          </div>
   391→        ) : (
   392→          <>
   393→            {/* Header pilih-semua dengan status indeterminate */}
   394→            <div className="flex items-center gap-3 bg-white rounded-xl border border-border px-4 py-3">
   395→              <input
   396→                type="checkbox"
   397→                checked={sel.allSelected}
   398→                ref={(el) => {
   399→                  if (el) el.indeterminate = sel.someSelected;
   400→                }}
   401→                onChange={sel.toggleAll}
   402→                aria-label="Pilih semua FAQ"
   403→                className="w-4 h-4 cursor-pointer"
   404→              />
   405→              <span className="text-sm font-medium text-suzuki-navy">
   406→                Pilih semua ({visible.length} FAQ)
   407→              </span>
   408→            </div>
   409→            <Accordion type="multiple" className="bg-white rounded-xl border border-border shadow-sm overflow-hidden">
   410→              {visible.map((f) => {
   411→                const idxGlobal = sortedAll.findIndex((x) => x.id === f.id);
   412→                const isSel = sel.selected.has(f.id);
   413→                return (
   414→                  <AccordionItem
   415→                    key={f.id}
   416→                    value={f.id}
   417→                    className={`border-border relative ${isSel ? "row-selected" : ""}`}
   418→                  >
   419→                    {/* Checkbox aksi massal — menumpuk di gutter kiri */}
   420→                    <div className="absolute left-3 sm:left-3.5 top-6 z-10">
   421→                      <input
   422→                        type="checkbox"
   423→                        checked={isSel}
   424→                        onChange={() => sel.toggle(f.id)}
   425→                        aria-label={`Pilih FAQ: ${f.pertanyaan.slice(0, 60)}`}
   426→                        className="w-4 h-4 cursor-pointer"
   427→                      />
   428→                    </div>
   429→                    <AccordionTrigger className="px-4 pl-10 sm:pl-11 sm:pr-5 py-4 hover:no-underline hover:bg-suzuki-red/5 transition-colors [&>svg]:hidden">
   430→                      <div className="flex items-start sm:items-center gap-3 flex-1 min-w-0 flex-col sm:flex-row">
   431→                        <span className="flex items-center gap-2 shrink-0">
   432→                          <span
   433→                            className="text-xs font-bold w-8 h-8 rounded-lg bg-muted text-suzuki-navy flex items-center justify-center"
   434→                            aria-hidden
   435→                            title={`Urutan ke-${f.urutan}`}
   436→                          >
   437→                            {f.urutan}
   438→                          </span>
   439→                          <span className={`text-[11px] px-2 py-0.5 rounded-full font-medium ${KATEGORI_WARNA[f.kategori] ?? "bg-muted text-muted-foreground"}`}>
   440→                            {FAQ_KATEGORI_LABEL[f.kategori] ?? f.kategori}
   441→                          </span>
   442→                          {!f.is_published && (
   443→                            <span className="text-[11px] px-2 py-0.5 rounded-full font-medium bg-amber-100 text-amber-700">
   444→                              Disembunyikan
   445→                            </span>
   446→                          )}
   447→                        </span>
   448→                        <span className="font-semibold text-suzuki-navy text-sm truncate text-left">
   449→                          {f.pertanyaan}
   450→                        </span>
   451→                      </div>
   452→                    </AccordionTrigger>
   453→                    <AccordionContent className="px-4 sm:px-5 pb-4 pt-0">
   454→                      <p className="text-sm text-muted-foreground leading-relaxed border-l-2 border-suzuki-red/30 pl-4 mb-4">
   455→                        {f.jawaban}
   456→                      </p>
   457→                      <div className="flex flex-wrap gap-2 pl-4">
   458→                        <button
   459→                          onClick={() => openEdit(f)}
   460→                          className="inline-flex items-center gap-1.5 px-3 py-1.5 border border-border text-suzuki-navy hover:bg-muted text-xs font-medium rounded-full transition-colors"
   461→                        >
   462→                          <Pencil className="w-3.5 h-3.5" aria-hidden />
   463→                          Edit
   464→                        </button>
   465→                        <button
   466→                          onClick={() => void onToggle(f)}
   467→                          className={`inline-flex items-center gap-1.5 px-3 py-1.5 text-xs font-medium rounded-full transition-colors ${
   468→                            f.is_published
   469→                              ? "border border-amber-200 text-amber-700 hover:bg-amber-50"
   470→                              : "bg-green-600 text-white hover:bg-green-700"
   471→                          }`}
   472→                        >
   473→                          {f.is_published ? (
   474→                            <>
   475→                              <EyeOff className="w-3.5 h-3.5" aria-hidden />
   476→                              Sembunyikan
   477→                            </>
   478→                          ) : (
   479→                            <>
   480→                              <Eye className="w-3.5 h-3.5" aria-hidden />
   481→                              Tampilkan
   482→                            </>
   483→                          )}
   484→                        </button>
